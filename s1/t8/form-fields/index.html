Subgrid — zarovnané sloupce, stylovatelný řádek

/* Rodič definuje 3 sloupce: label / input / hint */
.form-grid {
  display: grid;
  grid-template-columns: max-content 1fr max-content;
}

/* Řádek si přes subgrid „vypůjčí" sloupce rodiče —
   labely a hinty se zarovnají napříč všemi řádky */
.form-row {
  grid-column: span 3;
  display: grid;
  grid-template-columns: subgrid;
}

/* Bonus proti display: contents — řádek zůstává skutečným
   boxem, takže ho lze stylovat (hover, background, border).
   display: contents by obal řádku zahodil úplně. */
.form-row:hover { background: var(--surface-main-dim); }
